        - Declares a focusable button ref map (`itemRefs = useRef(new Map<string, HTMLButtonElement>())`) tracking primary slot selection button DOM elements, a fallback container ref (`listContainerRef = useRef<HTMLDivElement | null>(null)`), `emptyContainerRef = useRef<HTMLDivElement | null>(null)`, and `renameInputRef = useRef<HTMLInputElement | null>(null)`.
        - Renders each slot row `.save-file-item` as a semantic list item (`role="listitem"`) within the slot list container (`[role="list"]`). When `!isRenaming`, renders the primary slot selection button `<button type="button" className="save-file-item-main-btn">` with natural sequential `Tab` order and decoupled keyboard arrow traversal (`ArrowDown`, `ArrowUp`, `Home`, `End`) moving DOM focus without invoking `onSelectFile`; when `isRenaming`, conditionally renders the inline rename container `<div className="save-file-rename-container" data-interactive="true">`. Interactive inputs are never nested inside buttons.
        - Centralizes slot selection triggers through `handleSelectSlot(fileName)` that intercepts in-flight deletions, active saving, active renaming, and identical loading slot re-triggering (`if (isDeletingRef.current || deletingFileName !== null || isSaving || isCommittingRef.current || Boolean(renamingFileName) || fileName === loadingFileName) return;`), while permitting cross-slot preemption when switching to a different slot during load.
        - Slot row click handler `handleRowClick` uses Parent Target Filtering (`INTERACTIVE_DESCENDANT_SELECTOR = 'button, a, input, select, textarea, [role="button"], [data-interactive="true"]'`), preventing clicks inside action buttons, rename inputs, or popovers from triggering slot selection without calling `stopPropagation()`.
        - Implements inline rename state machine (`renamingFileName`, `draftName`, `renameError`) with three persistent concurrency latches (`isCommittingRef`, `isCancellingRef`, `isCommittedRef`) to serialize execution, eliminate `Enter` vs `blur` commit races, and prevent duplicate async I/O.
        - String comparisons and mutations enforce Unicode NFC normalization (`normalize('NFC')`).
        - Action buttons are globally disabled across all slot rows during active saving, renaming, or deletion (`disabled={Boolean(renamingFileName) || isLoading || Boolean(deletingFileName) || isSaving}`).
        - Deletion handler uses synchronous concurrency latching (`isDeletingRef = useRef(false)`), unique sentinel `EMPTY_LIST_SENTINEL = Symbol('EMPTY_LIST')` to avoid filename collision, wraps `onDeleteFile` in `try ... catch ... finally` to prevent unhandled promise rejections, and deterministically transfers focus via `useEffect([files])` to the surviving sibling slot button or empty container (`<div ref={emptyContainerRef} className="save-file-list-empty" role="status" tabIndex={-1}>`), eliminating focus drops to `document.body` (WCAG 2.4.3):
          ```tsx
          // src/renderer/save-editor/components/sidebar/SaveFileList.tsx
          import React, { useRef, useState, useEffect, useLayoutEffect } from 'react';
          import { sanitizeUserPaths } from '../../../../shared/path-sanitizer';
          import { SaveFileActions } from './SaveFileActions';

          export interface SaveFileListViewProps {
            files?: string[];
            activeFileName?: string | null;
            loadingFileName?: string | null;
            isSaving?: boolean;
            openPopoverFileName?: string | null;
            onOpenPopoverFileNameChange?: (fileName: string | null) => void;
            onSelectFile?: (fileName: string) => void;
            renderItemActions?: (fileName: string) => React.ReactNode;
            onDeleteFile?: (fileName: string) => Promise<void>;
            onRenameFile?: (fileName: string, nextName: string) => Promise<void>;
            listFocusRef?: React.RefObject<HTMLDivElement | null>;
          }

          const INTERACTIVE_DESCENDANT_SELECTOR = 'button, a, input, select, textarea, [role="button"], [data-interactive="true"]';
          const EMPTY_LIST_SENTINEL = Symbol('EMPTY_LIST');

          export function SaveFileListView({
            files = [],
            activeFileName = null,
            loadingFileName = null,
            isSaving = false,
            openPopoverFileName,
            onOpenPopoverFileNameChange,
            onSelectFile,
            renderItemActions,
            onDeleteFile,
            onRenameFile,
            listFocusRef,
          }: SaveFileListViewProps) {
            const [renamingFileName, setRenamingFileName] = useState<string | null>(null);
            const [draftName, setDraftName] = useState('');
            const [renameError, setRenameError] = useState<string | null>(null);
            const [deletingFileName, setDeletingFileName] = useState<string | null>(null);

            const itemRefs = useRef<Map<string, HTMLButtonElement>>(new Map());
            const listContainerRef = useRef<HTMLDivElement | null>(null);
            const emptyContainerRef = useRef<HTMLDivElement | null>(null);
            const renameInputRef = useRef<HTMLInputElement | null>(null);
            type PendingRenameFocus =
              | { type: 'input'; fileName: string }
              | {
                  type: 'item';
                  primaryFileName: string;
                  fallbackFileName: string;
                  fallbackFocused: boolean;
                };

            const pendingFocusTargetRef = useRef<string | typeof EMPTY_LIST_SENTINEL | null>(null);
            const pendingRenameFocusRef = useRef<PendingRenameFocus | null>(null);
            const restoreFocusOnCommitRef = useRef(true);
            const isDeletingRef = useRef(false);
            const isCommittingRef = useRef(false);
            const isCancellingRef = useRef(false);
            const isCommittedRef = useRef(false);

            useLayoutEffect(() => {
              if (renamingFileName !== null) {
                const pending = pendingRenameFocusRef.current;
                if (pending?.type === 'input' && renamingFileName === pending.fileName && renameInputRef.current) {
                  renameInputRef.current.focus();
                  const lastDotIndex = pending.fileName.lastIndexOf('.');
                  if (lastDotIndex > 0) {
                    renameInputRef.current.setSelectionRange(0, lastDotIndex);
                  } else {
                    renameInputRef.current.select();
                  }
                  pendingRenameFocusRef.current = null;
                }
                return;
              }

              // Always reset cancellation guard when exiting rename mode, even if restoreFocus was false
              isCancellingRef.current = false;
              const pending = pendingRenameFocusRef.current;
              if (pending?.type === 'item') {
                const primaryEl = itemRefs.current.get(pending.primaryFileName);
                const fallbackEl = itemRefs.current.get(pending.fallbackFileName);
                if (primaryEl) {
                  const activeEl = document.activeElement;
                  const canTransferFromFallback =
                    !pending.fallbackFocused ||
                    !activeEl ||
                    activeEl === document.body ||
                    !activeEl.isConnected ||
                    activeEl === fallbackEl;
                  if (canTransferFromFallback) {
                    primaryEl.focus();
                  }
                  pendingRenameFocusRef.current = null;
                } else if (!pending.fallbackFocused) {
                  fallbackEl?.focus();
                  if (
                    pending.primaryFileName !== pending.fallbackFileName &&
                    files.includes(pending.fallbackFileName)
                  ) {
                    pending.fallbackFocused = true;
                  } else {
                    pendingRenameFocusRef.current = null;
                  }
                } else if (!files.includes(pending.fallbackFileName)) {
                  pendingRenameFocusRef.current = null;
                }
              }
            }, [renamingFileName, files]);

            const startRename = (fileName: string) => {
              if (isCommittingRef.current || renamingFileName !== null || isDeletingRef.current || deletingFileName !== null) return;
              isCancellingRef.current = false;
              isCommittedRef.current = false;
              pendingRenameFocusRef.current = { type: 'input', fileName };
              setRenamingFileName(fileName);
              setDraftName(fileName);
              setRenameError(null);
            };

            const cancelRename = (
              fileName: string,
              options: { restoreFocus?: boolean } = { restoreFocus: true }
            ) => {
              if (isCancellingRef.current || isCommittingRef.current) return;
              const { restoreFocus = true } = options;
              isCancellingRef.current = true;
              pendingRenameFocusRef.current = restoreFocus
                ? {
                    type: 'item',
                    primaryFileName: fileName,
                    fallbackFileName: fileName,
                    fallbackFocused: false,
                  }
                : null;
              setRenamingFileName(null);
              setRenameError(null);
            };

            const commitRename = async (
              fileName: string,
              options: { restoreFocus?: boolean } = { restoreFocus: true }
            ) => {
              if (isCommittingRef.current || isCancellingRef.current || isCommittedRef.current) return;
              const { restoreFocus = true } = options;
              const normalizedDraft = draftName.trim().normalize('NFC');
              const normalizedFile = fileName.normalize('NFC');

              if (!normalizedDraft) {
                setRenameError('File name cannot be empty');
                renameInputRef.current?.focus();
                return;
              }
              if (/[\\/:*?"<>|]/.test(normalizedDraft)) {
                setRenameError('Name contains invalid characters');
                renameInputRef.current?.focus();
                return;
              }
              if (normalizedDraft === normalizedFile) {
                cancelRename(fileName, { restoreFocus });
                return;
              }

              isCommittingRef.current = true;
              restoreFocusOnCommitRef.current = restoreFocus;
              try {
                await onRenameFile?.(fileName, normalizedDraft);
                isCommittedRef.current = true;
                pendingRenameFocusRef.current = restoreFocusOnCommitRef.current
                  ? {
                      type: 'item',
                      primaryFileName: normalizedDraft,
                      fallbackFileName: fileName,
                      fallbackFocused: false,
                    }
                  : null;
                setRenamingFileName((prev) => (prev === fileName ? null : prev));
                setRenameError(null);
              } catch (err: any) {
                setRenameError(sanitizeUserPaths(err?.message || 'Failed to rename save file'));
                renameInputRef.current?.focus();
              } finally {
                isCommittingRef.current = false;
              }
            };

            const handleDelete = async (fileName: string) => {
              if (isDeletingRef.current || deletingFileName !== null) return;
              isDeletingRef.current = true;

              const idx = files.indexOf(fileName);
              if (idx === -1) {
                isDeletingRef.current = false;
                return;
              }
              const nextTargetName = files[idx + 1] ?? files[idx - 1] ?? EMPTY_LIST_SENTINEL;

              pendingFocusTargetRef.current = nextTargetName;
              pendingRenameFocusRef.current = null;
              setDeletingFileName(fileName);

              try {
                await onDeleteFile?.(fileName);
              } catch (error) {
                const rawMsg = error instanceof Error ? error.message : String(error);
                console.error('[SaveFileListView] Failed to delete save file:', {
                  fileName: sanitizeUserPaths(fileName),
                  error: sanitizeUserPaths(rawMsg),
                  stack: error instanceof Error && error.stack ? sanitizeUserPaths(error.stack) : undefined,
                });
                pendingFocusTargetRef.current = null;
                throw error;
              } finally {
                isDeletingRef.current = false;
                setDeletingFileName(null);
              }
            };

            const handleSelectSlot = (fileName: string) => {
              if (
                isDeletingRef.current ||
                deletingFileName !== null ||
                isSaving ||
                isCommittingRef.current ||
                Boolean(renamingFileName) ||
                fileName === loadingFileName
              ) {
                return;
              }
              onSelectFile?.(fileName);
            };

            const handleButtonKeyDown = (e: React.KeyboardEvent<HTMLButtonElement>, fileName: string) => {
              if (e.defaultPrevented) return;

              const currentIndex = files.indexOf(fileName);
              if (currentIndex === -1) return;

              let targetFile: string | undefined;

              if (e.key === 'ArrowDown') {
                e.preventDefault();
                targetFile = files[Math.min(currentIndex + 1, files.length - 1)];
              } else if (e.key === 'ArrowUp') {
                e.preventDefault();
                targetFile = files[Math.max(currentIndex - 1, 0)];
              } else if (e.key === 'Home') {
                e.preventDefault();
                targetFile = files[0];
              } else if (e.key === 'End') {
                e.preventDefault();
                targetFile = files[files.length - 1];
              }

              if (targetFile) {
                itemRefs.current.get(targetFile)?.focus();
              }
            };

            const handleRowClick = (e: React.MouseEvent<HTMLDivElement>, fileName: string) => {
              if (e.defaultPrevented) return;
              // Prevent row selection if click occurred inside buttons, inputs, actions, or popovers
              if ((e.target as HTMLElement).closest(INTERACTIVE_DESCENDANT_SELECTOR)) return;
              itemRefs.current.get(fileName)?.focus();
              handleSelectSlot(fileName);
            };

            // Deterministic post-render focus retention targeting focusable button elements
            useEffect(() => {
              if (!pendingFocusTargetRef.current) return;

              if (pendingFocusTargetRef.current === EMPTY_LIST_SENTINEL) {
                emptyContainerRef.current?.focus();
              } else {
                const targetBtn = itemRefs.current.get(pendingFocusTargetRef.current);
                if (targetBtn) {
                  targetBtn.focus();
                } else if (files.length > 0 && files[0] && itemRefs.current.get(files[0])) {
                  itemRefs.current.get(files[0])?.focus();
                } else {
                  listContainerRef.current?.focus();
                }
              }
              pendingFocusTargetRef.current = null;
            }, [files]);

            if (files.length === 0) {
              return (
                <div
                  ref={(el) => {
                    emptyContainerRef.current = el;
                    if (listFocusRef) {
                      (listFocusRef as React.MutableRefObject<HTMLDivElement | null>).current = el;
                    }
                  }}
                  className="save-file-list-empty"
                  role="status"
                  aria-live="polite"
                  tabIndex={-1}
                  style={{ outline: 'none' }}
                >
                  <span>No save files found in directory</span>
                </div>
              );
            }

            return (
              <div
                ref={(el) => {
                  listContainerRef.current = el;
                  if (listFocusRef) {
                    (listFocusRef as React.MutableRefObject<HTMLDivElement | null>).current = el;
                  }
                }}
                className="save-file-list"
                role="list"
                aria-label="Save file slots"
                data-testid="save-file-list"
                tabIndex={-1}
              >
                {files.map((fileName) => {
                  const isActive = fileName === activeFileName;
                  const isLoading = fileName === loadingFileName;
                  const isRenaming = fileName === renamingFileName;

                  return (
                    <div
                      key={fileName}
                      role="listitem"
                      className={`save-file-item ${isActive ? 'active' : ''} ${isLoading ? 'loading' : ''} ${openPopoverFileName === fileName ? 'has-open-popover' : ''}`.trim()}
                      onClick={(e) => handleRowClick(e, fileName)}
                    >
                      {isRenaming ? (
                        <div className="save-file-rename-container" data-interactive="true">
                          <input
                            ref={renameInputRef}
                            className="save-file-rename-input"
                            value={draftName}
                            onChange={(e) => {
                              setDraftName(e.target.value);
                              if (renameError) setRenameError(null);
                            }}
                            onKeyDown={(e) => {
                              if (e.key === 'Enter') {
                                e.preventDefault();
                                commitRename(fileName, { restoreFocus: true });
                              } else if (e.key === 'Escape') {
                                e.preventDefault();
                                cancelRename(fileName, { restoreFocus: true });
                              }
                            }}
                            onBlur={(e) => {
                              if (isCommittingRef.current) {
                                if (e.relatedTarget) {
                                  restoreFocusOnCommitRef.current = false;
                                }
                                return;
                              }
                              if (isCancellingRef.current || isCommittedRef.current) {
                                if (e.relatedTarget) {
                                  pendingRenameFocusRef.current = null;
                                }
                                return;
                              }
                              commitRename(fileName, { restoreFocus: !e.relatedTarget });
                            }}
                          />
                          {renameError && (
                            <div className="save-editor-rename-tooltip" role="alert">
                              {renameError}
                            </div>
                          )}
                        </div>
                      ) : (
                        <button
                          type="button"
                          ref={(el) => {
                            if (el) itemRefs.current.set(fileName, el);
                            else itemRefs.current.delete(fileName);
                          }}
                          className="save-file-item-main-btn"
                          onClick={() => handleSelectSlot(fileName)}
                          onKeyDown={(e) => handleButtonKeyDown(e, fileName)}
                          aria-current={isActive ? 'true' : undefined}
                          aria-busy={isLoading ? 'true' : undefined}
                          aria-disabled={
                            Boolean(deletingFileName) ||
                            Boolean(renamingFileName) ||
                            fileName === loadingFileName ||
                            isSaving
                              ? 'true'
                              : undefined
                          }
                        >
                          <span className="save-file-name">{fileName}</span>
                        </button>
                      )}

                      {renderItemActions ? (
                        <div className="save-file-actions" data-interactive="true">
                          {renderItemActions(fileName)}
                        </div>
                      ) : (
                        <SaveFileActions
                          fileName={fileName}
                          onStartRename={() => startRename(fileName)}
                          onDelete={() => handleDelete(fileName)}
                          disabled={Boolean(renamingFileName) || isLoading || Boolean(deletingFileName) || isSaving}
                          isConfirmOpen={
                            openPopoverFileName !== undefined && onOpenPopoverFileNameChange !== undefined
                              ? openPopoverFileName === fileName
                              : undefined
                          }
                          onConfirmOpenChange={
                            openPopoverFileName !== undefined && onOpenPopoverFileNameChange !== undefined
                              ? (open) => onOpenPopoverFileNameChange(open ? fileName : null)
                              : undefined
                          }
                        />
                      )}
                    </div>
                  );
                })}
              </div>
            );
          }
          ```

          ```tsx
       }
          ```

     ```typescript
         activeUnsubscribeRef.current?.();

       }

       currentLoadTokenRef.current += 1;
       abortInFlightLoad();
     };

       return () => {
         invalidatePendingLoads();
       };
     }, [gameKey, electronAPI]);

         return;

           activeLoadingSlotRef.current = null;
         }
       }
     ```

### [NEW] [`src/renderer/utils/errors.ts`](file:///d:/Projects/YumeShelf/src/renderer/utils/errors.ts)
- Common error classification utilities including `isCancellationError(err: unknown): boolean` recognizing explicit user cancellations and abort signals while avoiding ambient fault masking of system/crypto errors:
  ```typescript
  /**
   * Determines whether an error represents an explicit user cancellation or abort signal.
   * Avoids ambient fault masking by strictly requiring cancellation-specific names, codes,
   * flags, exact "aborted" matching, or explicit user/operation cancellation phrases,
   * banning bare substring "aborted" matching that could mask system/crypto failures.
   */
  export function isCancellationError(err: unknown): boolean {
    if (err === null || err === undefined) {
      return false;
    }
    if (typeof err === 'object') {
      const errorObj = err as Record<string, unknown>;
      if (
        errorObj.name === 'AbortError' ||
        errorObj.name === 'CanceledError' ||
        errorObj.code === 'ERR_CANCELED' ||
        errorObj.isCancelled === true ||
        errorObj.isCanceled === true
      ) {
        return true;
      }
      if (typeof errorObj.message === 'string') {
        const msg = errorObj.message.trim();
        if (msg.toLowerCase() === 'aborted') {
          return true;
        }
        return (
          /\buser\s+cancell?ed\b/i.test(msg) ||
          /\boperation\s+cancell?ed\b/i.test(msg) ||
          /\b(cancelled|canceled)\b/i.test(msg) ||
          /\b(user|operation)\s+aborted\b/i.test(msg) ||
          /\b(the\s+)?operation\s+was\s+aborted\b/i.test(msg) ||
          /\b(the\s+)?user\s+aborted\b/i.test(msg) ||
          /\baborted\s+by\s+user\b/i.test(msg)
        );
      }
    }
    if (typeof err === 'string') {
      const str = err.trim();
      if (str.toLowerCase() === 'aborted') {
        return true;
      }
      return (
        /\buser\s+cancell?ed\b/i.test(str) ||
        /\boperation\s+cancell?ed\b/i.test(str) ||
        /\b(cancelled|canceled)\b/i.test(str) ||
        /\b(user|operation)\s+aborted\b/i.test(str) ||
        /\b(the\s+)?operation\s+was\s+aborted\b/i.test(str) ||
        /\b(the\s+)?user\s+aborted\b/i.test(str) ||
        /\baborted\s+by\s+user\b/i.test(str)
      );
    }
    return false;
  }
  ```
