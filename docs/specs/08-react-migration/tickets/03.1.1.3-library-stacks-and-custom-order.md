       ```typescript
       export function readStoredCustomOrder(
         storage: UiStorageAdapter = defaultStorage,
         nowProvider: () => number = Date.now
       ): string[] {
         let raw: string | null = null;
         try {
           raw = storage.getItem('yumeshelf_custom_order');
           if (!raw) return [];
           const parsed = JSON.parse(raw);
           if (parsed && typeof parsed === 'object') {
             if (typeof (parsed as any).schemaVersion === 'number' && (parsed as any).schemaVersion > 1) {
               // Forward schema version: return empty array to avoid mutating unknown structure
               return [];
             }
             if ((parsed as any).schemaVersion === 1 && Array.isArray((parsed as any).order)) {
               return (parsed as any).order.filter((item: any): item is string => typeof item === 'string' && item.trim().length > 0);
             }
             // Defense-in-depth: support legacy raw string array if accessed outside startup migrateUiPreferences
             if (Array.isArray(parsed)) {
               return parsed.filter((item: any): item is string => typeof item === 'string' && item.trim().length > 0);
             }
           }
           // Non-object or unrecognized malformed structure triggers quarantine recovery
           throw new SyntaxError('Stored custom order payload is malformed or invalid schema structure');
         } catch (err) {
           if (err instanceof SyntaxError && raw !== null) {
             console.warn('[LIBRARY-ORDER] Corrupt custom order payload encountered; quarantining:', formatSanitizedError(err));
             const corruptKey = `yumeshelf_custom_order_corrupt_${nowProvider()}`;
             try {
               storage.setItem(corruptKey, raw);
             } catch (storageErr) {
               console.warn('[LIBRARY-ORDER] Failed during corrupt custom order recovery:', formatSanitizedError(storageErr));
             } finally {
               try {
                 storage.removeItem('yumeshelf_custom_order');
               } catch (removeErr) {
                 console.warn('[LIBRARY-ORDER] Failed to remove corrupt custom order key:', formatSanitizedError(removeErr));
               }
             }
           } else {
             console.warn('[LIBRARY-ORDER] Failed to read custom order from storage:', formatSanitizedError(err));
           }
           return [];
         }
       }
       ```

     - Extract and export a pure function `resolveCustomOrder(games: readonly GameEntry[], storedOrder: readonly string[] = readStoredCustomOrder()): string[]` that accepts an optional `storedOrder` defaulting to `readStoredCustomOrder()` and computes the re-indexed custom order purely in memory without invoking storage setters.
     - **Reconciliation Algorithm Contract**:
       `resolveCustomOrder` executes a deterministic multi-phase reconciliation purely in memory:
       1. Resolves legacy pre-Epic 07 aliases (`folderName`, `gameId`, `primaryInstance.gameKey`) to authoritative `gameKey` identifiers.
       2. Discards obsolete, unresolvable, or deleted game entries while preserving user-defined relative order.
       3. Deduplicates entries (`new Set<string>()`).
       4. Deterministically appends newly discovered games present in `games` but absent in `storedOrder` to the end of the returned array.
       ```typescript
       export function resolveCustomOrder(
         games: readonly GameEntry[],
         storedOrder: readonly string[] = readStoredCustomOrder()
       ): string[] {
         const rawOrder = Array.isArray(storedOrder) ? storedOrder : readStoredCustomOrder();
         const keySet = new Set(games.map((g) => getGameKey(g)).filter(Boolean));
         const aliasMatches = new Map<string, string[]>();

         for (const game of games) {
           const primaryKey = getGameKey(game);
           if (!primaryKey) continue;

           const aliases = [
             String(game.gameKey || '').trim(),
             String(game.gameId || '').trim(),
             String(game.folderName || '').trim(),
             String(game.primaryInstance?.gameKey || '').trim()
           ].filter(Boolean);

           for (const alias of aliases) {
             const matches = aliasMatches.get(alias) || [];
             if (!matches.includes(primaryKey)) {
               matches.push(primaryKey);
             }
             aliasMatches.set(alias, matches);
           }
         }

         const migratedOrder: string[] = [];
         for (const entry of rawOrder) {
           if (typeof entry !== 'string') continue;
           const trimmed = entry.trim();
           if (!trimmed) continue;

           if (keySet.has(trimmed)) {
             migratedOrder.push(trimmed);
             continue;
           }

           const matches = aliasMatches.get(trimmed) || [];
           if (matches.length === 1) {
             migratedOrder.push(matches[0]);
           }
         }

         const nextOrder: string[] = [];
         const seen = new Set<string>();

         for (const key of migratedOrder) {
           if (keySet.has(key) && !seen.has(key)) {
             nextOrder.push(key);
             seen.add(key);
           }
         }

         for (const game of games) {
           const key = getGameKey(game);
           if (key && !seen.has(key)) {
             nextOrder.push(key);
             seen.add(key);
           }
         }

         return nextOrder;
       }

       export function areStringArraysEqual(
         a?: readonly string[] | null,
         b?: readonly string[] | null
       ): boolean {
         if (a === b) return true;
         const aLen = a?.length ?? 0;
         const bLen = b?.length ?? 0;
         if (aLen === 0 && bLen === 0) return true;
         if (aLen !== bLen || !a || !b) return false;
         for (let i = 0; i < aLen; i++) {
           if (a[i] !== b[i]) return false;
         }
         return true;
       }

       export function writeCustomOrder(
         order: readonly string[],
         storage: UiStorageAdapter = defaultStorage
       ): void {
         try {
           // Forward schema defense: probe existing storage before writing
           const existingRaw = storage.getItem('yumeshelf_custom_order');
           if (existingRaw) {
             try {
               const existing = JSON.parse(existingRaw);
               if (existing && typeof existing === 'object' && typeof existing.schemaVersion === 'number' && existing.schemaVersion > 1) {
                 console.warn('[LIBRARY-ORDER] Aborting write: custom order schema version is newer than supported version:', {
                   existingVersion: existing.schemaVersion,
                   maxSupportedVersion: 1,
                   internal_outcome: 'ABORTED',
                   abort_reason: 'FUTURE_SCHEMA_VERSION'
                 });
                 return;
               }
             } catch {
               // SyntaxError in existing value is ignored here (readStoredCustomOrder handles quarantine)
             }
           }
           const sanitized = Array.isArray(order)
             ? order.filter((item): item is string => typeof item === 'string' && item.trim().length > 0)
             : [];
           storage.setItem('yumeshelf_custom_order', JSON.stringify({ schemaVersion: 1, order: sanitized }));
         } catch (err) {
           console.warn('[LIBRARY-ORDER] Failed to persist custom order:', formatSanitizedError(err));
         }
       }

       export function migrateStoredCustomOrderAliases(
         games: readonly GameEntry[],
         storedOrder: readonly string[]
       ): string[] | null {
         if (!Array.isArray(storedOrder) || storedOrder.length === 0 || !Array.isArray(games) || games.length === 0) {
           return null;
         }

         const keySet = new Set(games.map((g) => getGameKey(g)).filter(Boolean));
         const aliasMatches = new Map<string, string[]>();

         for (const game of games) {
           const primaryKey = getGameKey(game);
           if (!primaryKey) continue;

           const aliases = [
             String(game.gameKey || '').trim(),
             String(game.gameId || '').trim(),
             String(game.folderName || '').trim(),
             String(game.primaryInstance?.gameKey || '').trim(),
           ].filter(Boolean);

           for (const alias of aliases) {
             if (alias === primaryKey) continue;
             const matches = aliasMatches.get(alias) || [];
             if (!matches.includes(primaryKey)) {
               matches.push(primaryKey);
             }
             aliasMatches.set(alias, matches);
           }
         }

         let didMutate = false;
         const migrated: string[] = [];
         const seen = new Set<string>();

         for (const entry of storedOrder) {
           if (typeof entry !== 'string') continue;
           const trimmed = entry.trim();
           if (!trimmed) continue;

           let targetKey = trimmed;
           // 1. Canonical Key Precedence: If keySet has trimmed, strictly preserve as canonical key; NEVER remap to another game's alias
           if (keySet.has(trimmed)) {
             targetKey = trimmed;
           } else {
             // 2. Ambiguity Elimination: Remap ONLY if matches.length === 1 (unambiguous 1:1 match)
             const matches = aliasMatches.get(trimmed) || [];
             if (matches.length === 1) {
               targetKey = matches[0];
               didMutate = true;
             }
             // Ambiguous matches (> 1) and unknown keys (0, such as disconnected drives) MUST be preserved intact without modification
           }

           if (!seen.has(targetKey)) {
             migrated.push(targetKey);
             seen.add(targetKey);
           } else {
             didMutate = true;
           }
         }

         return didMutate || !areStringArraysEqual(storedOrder, migrated) ? migrated : null;
       }
       ```

       ```typescript
       export function resolveStackByGroupKey(
         games: readonly GameEntry[],
         groupKey: string
       ): StackLibraryViewItem | null {
         if (!groupKey) return null;

         const targetGame = games.find((g) =>
           g.gameId === groupKey ||
           g.gameKey === groupKey ||
           buildGroupKey(g) === groupKey ||
           (Array.isArray(g.instances) && g.instances.some((i) => i && (i.gameKey === groupKey || i.instanceId === groupKey)))
         );
         if (!targetGame) return null;

         const rawInstances = Array.isArray(targetGame.instances) ? targetGame.instances : [targetGame];
         if (rawInstances.length <= 1) return null;

         const orderedGames = [...rawInstances].sort((a, b) => {
           const depthA = (a.relativePath || '').split(/[\\/]+/).filter(Boolean).length;
           const depthB = (b.relativePath || '').split(/[\\/]+/).filter(Boolean).length;
           if (depthA !== depthB) return depthA - depthB;
           return (a.relativePath || '').localeCompare(b.relativePath || '');
         });

         const recents = orderedGames
           .filter((g) => (g.lastPlayed || 0) > 0)
           .sort((a, b) => (b.lastPlayed || 0) - (a.lastPlayed || 0));
         const primaryGame = targetGame.primaryInstance || (recents.length > 0 ? recents[0] : orderedGames[0]);

         return {
           type: 'stack',
           isStack: true,
           groupKey: buildGroupKey(targetGame),
           favorite: !!targetGame.favorite,
           primaryGame,
           games: orderedGames,
           representativeKey: getGameKey(targetGame),
           stackSize: orderedGames.length,
         };
       }
       ```
       Performs a fast linear lookup for the target group and sorts only its child instances ($O(K \log K)$ where $K$ is the small instance count of the target stack), eliminating $O(N \log N)$ sorting and mass object allocations across thousands of unrelated games.

        ```typescript
        export function patchRunningGameViewItem(
          prevItems: readonly LibraryViewItem[],
          changedGame: GameEntry
        ): readonly LibraryViewItem[] {
          if (!changedGame) return prevItems;
          const changedKey = getGameKey(changedGame);
          if (!changedKey) return prevItems;
          const changedGroupKey = buildGroupKey(changedGame);
          let didMutate = false;

          const nextItems = prevItems.map((item) => {
            const matchesRepresentative = item.representativeKey === changedKey;
            const matchesGroup = item.groupKey === changedGroupKey;
            const matchesAnyChild = item.isStack && item.games.some((g) => getGameKey(g) === changedKey);

            if (!matchesRepresentative && !matchesGroup && !matchesAnyChild) {
              return item;
            }

            didMutate = true;
            if (item.type === 'single') {
              return {
                ...item,
                primaryGame: changedGame,
                games: [changedGame] as const,
              };
            } else {
              const instanceMap = Array.isArray(changedGame.instances) && changedGame.instances.length > 0
                ? new Map(changedGame.instances.map((inst) => [getGameKey(inst), inst]))
                : null;

              const nextGames = instanceMap
                ? item.games.map((g) => instanceMap.get(getGameKey(g)) ?? g)
                : item.games.map((g) => (getGameKey(g) === changedKey ? changedGame : g));

              const recents = nextGames
                .filter((g) => (g.lastPlayed || 0) > 0)
                .sort((a, b) => (b.lastPlayed || 0) - (a.lastPlayed || 0));

              const explicitPrimaryKey = changedGame.primaryInstance
                ? getGameKey(changedGame.primaryInstance)
                : null;

              const primaryGame =
                (explicitPrimaryKey ? nextGames.find((g) => getGameKey(g) === explicitPrimaryKey) : null) ??
                (recents.length > 0 ? recents[0] : null) ??
                (item.primaryGame ? nextGames.find((g) => getGameKey(g) === getGameKey(item.primaryGame)) : null) ??
                nextGames[0];

              return {
                ...item,
                primaryGame,
                games: nextGames,
                stackSize: nextGames.length,
              };
            }
          });

          return didMutate ? nextItems : prevItems;
        }
        ```
        Reconciles in-flight runtime status changes on individual games while preserving referential identity (`Object.is(prevItems[i], nextItems[i]) === true`) for all untouched items.
