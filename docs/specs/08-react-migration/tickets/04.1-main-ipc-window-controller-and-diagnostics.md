     ```typescript
     export class CircularRingBuffer<T> {
       private buffer: (T | undefined)[];
       private head = 0;
       private size = 0;
       private droppedCount = 0;

       constructor(readonly capacity: number) {
         if (!Number.isInteger(capacity) || capacity <= 0) {
           throw new Error('Capacity must be a positive integer');
         }
         this.buffer = new Array(capacity);
       }

       push(item: T): void {
         let entry = item;
         if (typeof item === 'string') {
           const stripped = item.replace(/\u001b\[[0-9;]*m/g, '');
           entry = (stripped.length > 2000 ? (stripped.slice(0, 2000) + '... [TRUNCATED]') : stripped) as unknown as T;
         }
         if (this.size === this.capacity) {
           this.droppedCount++;
         }
         this.buffer[this.head] = entry;
         this.head = (this.head + 1) % this.capacity;
         if (this.size < this.capacity) this.size++;
       }

       getDroppedCount(): number {
         return this.droppedCount;
       }

       toArray(): T[] {
         const result: T[] = [];
         const start = this.size < this.capacity ? 0 : this.head;
         for (let i = 0; i < this.size; i++) {
           result.push(this.buffer[(start + i) % this.capacity] as T);
         }
         return result;
       }

       toSafeArray(maxDepth = 10): unknown[] {
         const DANGEROUS_KEYS = new Set(['__proto__', 'constructor', 'prototype']);
         const ANSI_REGEX = /\u001b\[[0-9;]*m/g;
         const sanitize = (val: unknown, activeStack = new Set<object>(), depth = 0): unknown => {
           if (depth > maxDepth) return '[DepthExceeded]';
           if (typeof val === 'string') {
             const stripped = val.replace(ANSI_REGEX, '');
             return stripped.length > 2000 ? `${stripped.slice(0, 2000)}... [truncated]` : stripped;
           }
           if (val !== null && typeof val === 'object') {
             if (activeStack.has(val)) return '[Circular]';
             activeStack.add(val);
             try {
               if (val instanceof Error) {
                 const errOut: Record<string, unknown> = Object.create(null);
                 errOut.name = val.name;
                 errOut.message = typeof val.message === 'string' ? sanitize(val.message, activeStack, depth + 1) : val.message;
                 errOut.stack = typeof val.stack === 'string' ? sanitize(val.stack, activeStack, depth + 1) : val.stack;
                 if ('cause' in val && val.cause !== undefined) {
                   errOut.cause = sanitize(val.cause, activeStack, depth + 1);
                 }
                 if (typeof AggregateError !== 'undefined' && val instanceof AggregateError && Array.isArray(val.errors)) {
                   errOut.errors = val.errors.map((e) => sanitize(e, activeStack, depth + 1));
                 }
                 return errOut;
               }
               if (val instanceof Date) {
                 return val.toISOString();
               }
               if (val instanceof Map) {
                 const mapOut: Record<string, unknown> = Object.create(null);
                 for (const [k, v] of val.entries()) {
                   const keyStr = typeof k === 'string' ? k : String(k);
                   if (!DANGEROUS_KEYS.has(keyStr)) {
                     mapOut[keyStr] = sanitize(v, activeStack, depth + 1);
                   }
                 }
                 return mapOut;
               }
               if (val instanceof Set) {
                 return Array.from(val).map((item) => sanitize(item, activeStack, depth + 1));
               }
               if (Array.isArray(val)) {
                 return val.map((item) => sanitize(item, activeStack, depth + 1));
               }
               const out: Record<string, unknown> = Object.create(null);
               for (const [k, v] of Object.entries(val)) {
                 if (!DANGEROUS_KEYS.has(k)) {
                   out[k] = sanitize(v, activeStack, depth + 1);
                 }
               }
               return out;
             } finally {
               activeStack.delete(val);
             }
           }
           if (typeof val === 'bigint') {
             return val.toString();
           }
           return val;
         };
         return this.toArray().map((entry) => sanitize(entry, new Set<object>(), 0));
       }

       toJSON(): unknown[] {
         return this.toSafeArray();
       }

       clear(): void {
         this.head = 0;
         this.size = 0;
         this.droppedCount = 0;
         this.buffer.fill(undefined);
       }
     }
     ```

     ```typescript
     export interface SanitizedErrorRecord {
       name?: string;
       message: string;
       stack?: string;
     }

     export function extractSanitizedCauseChain(err: unknown, maxDepth = 5): SanitizedErrorRecord[] {
       const chain: SanitizedErrorRecord[] = [];
       let current: unknown = err;
       let depth = 0;
       const seen = new Set<unknown>();

       while (current !== null && current !== undefined && depth < maxDepth) {
         if (seen.has(current)) {
           break;
         }
         seen.add(current);

         if (current instanceof Error) {
           chain.push({
             name: current.name,
             message: sanitizeUserPaths(current.message || 'Unknown error'),
             stack: current.stack ? sanitizeUserPaths(current.stack) : undefined,
           });
           current = (current as { cause?: unknown }).cause;
         } else {
           chain.push({
             message: sanitizeUserPaths(current) || 'Unknown error',
           });
           break;
         }
         depth++;
       }
       return chain;
     }

     export function registerGlobalErrorSanitizer(
       targetWindow: Window | any = typeof window !== 'undefined' ? window : undefined,
       prefix: string = '[Renderer]'
     ): () => void {
       if (!targetWindow || typeof targetWindow.addEventListener !== 'function') {
         return () => {};
       }

       const errorHandler = (event: any) => {
         const rawError = event.error;
         const sanitizedError = rawError instanceof Error
           ? {
               name: rawError.name,
               message: sanitizeUserPaths(rawError.message || ''),
               stack: rawError.stack ? sanitizeUserPaths(rawError.stack) : undefined,
               causeChain: rawError.cause ? extractSanitizedCauseChain(rawError.cause) : undefined,
             }
           : typeof rawError === 'string'
             ? sanitizeUserPaths(rawError)
             : rawError;

         console.error(`${prefix} Uncaught error:`, {
           message: sanitizeUserPaths(event.message || ''),
           filename: event.filename ? sanitizeUserPaths(event.filename) : '',
           lineno: event.lineno,
           colno: event.colno,
           error: sanitizedError,
         });
       };

       const unhandledRejectionHandler = (event: any) => {
         const reason = event.reason;
         const sanitizedReason = reason instanceof Error
           ? {
               name: reason.name,
               message: sanitizeUserPaths(reason.message || ''),
               stack: reason.stack ? sanitizeUserPaths(reason.stack) : undefined,
               causeChain: reason.cause ? extractSanitizedCauseChain(reason.cause) : undefined,
             }
           : typeof reason === 'string'
             ? sanitizeUserPaths(reason)
             : reason;

         console.error(`${prefix} Unhandled rejection:`, {
           reason: sanitizedReason,
         });
       };

       targetWindow.addEventListener('error', errorHandler);
       targetWindow.addEventListener('unhandledrejection', unhandledRejectionHandler);

       return () => {
         targetWindow.removeEventListener('error', errorHandler);
         targetWindow.removeEventListener('unhandledrejection', unhandledRejectionHandler);
       };
     }
     ```

         if (isQuitting) {
           return;
         }

         // 2. Clean exit handling
         if (reason === 'clean-exit') {
           app.quit();
           return;
         }

         // 3. Unexpected crash handling
         destroyTrayIcon();
         mainWindow = null;
         try {
           if (!win.isDestroyed()) {
             win.destroy();
           }
         } catch (destroyErr) {
           console.error('[MainWindow] Error destroying crashed window:', {
             error: sanitizeUserPaths(destroyErr instanceof Error ? destroyErr.message : destroyErr),
             stack: destroyErr instanceof Error && destroyErr.stack ? sanitizeUserPaths(destroyErr.stack) : undefined,
           });
         }

         try {
           const choice = dialog.showMessageBoxSync({
             type: 'error',
             title: 'YumeShelf Renderer Crash',
             message: 'The YumeShelf display process terminated unexpectedly.',
             detail: `Reason: ${reason} (Exit Code: ${exitCode})\n\nWould you like to relaunch the application or exit?`,

             defaultId: 0,
             cancelId: 1,
             noLink: true
           });

             app.relaunch();
             app.exit(0);
           } else {
             app.quit();
           }
         } catch (dialogErr) {
           console.error('[MainWindow] Failed to display crash recovery dialog:', {
             error: sanitizeUserPaths(dialogErr instanceof Error ? dialogErr.message : dialogErr),
             stack: dialogErr instanceof Error && dialogErr.stack ? sanitizeUserPaths(dialogErr.stack) : undefined,
           });
           app.quit();
         }
       });
