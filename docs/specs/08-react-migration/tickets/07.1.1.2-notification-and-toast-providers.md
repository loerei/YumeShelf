       ```typescript
       export type AppUpdatePhase = 'available' | 'downloading' | 'ready' | 'error';

       export type AppUpdateNotification =
         | {
             readonly phase: 'available';
             readonly releaseName?: string;
             readonly releaseNotes?: string;
             readonly releaseUrl?: string;
             readonly downloadable?: boolean;
           }
         | {
             readonly phase: 'downloading';
             readonly releaseName?: string;
             readonly releaseNotes?: string;
             readonly releaseUrl?: string;
             readonly progress?: number;
           }
         | {
             readonly phase: 'ready';
             readonly releaseName?: string;
             readonly releaseNotes?: string;
             readonly releaseUrl?: string;
           }
         | {
             readonly phase: 'error';
             readonly releaseName?: string;
             readonly releaseNotes?: string;
             readonly releaseUrl?: string;
             readonly error: string;
           };

       export interface RawAppUpdatePayload {
         phase:
           | 'update-available'
           | 'download-started'
           | 'download-progress'
           | 'download-ready'
           | 'download-failed'
           | 'install-preparing'
           | 'install-deferred'
           | 'install-handoff';
         update?: {
           releaseName?: string;
           releaseNotes?: string;
           releaseUrl?: string;
           downloadable?: boolean;
           downloadReady?: boolean;
         };
         progress?: number;
         downloaded?: number;
         total?: number;
         error?: string;
         reason?: string;
       }

       export function clampProgress(val: unknown): number | undefined {
         if (typeof val !== 'number' || !Number.isFinite(val)) return undefined;
         return Math.max(0, Math.min(1, val));
       }

       export function parseAppUpdatePayload(raw: unknown): AppUpdateNotification | null {
         if (!raw || typeof raw !== 'object' || !('phase' in raw)) return null;
         const payload = raw as RawAppUpdatePayload;
         const update = payload.update;
         switch (payload.phase) {
           case 'update-available':
             return {
               phase: 'available',
               releaseName: update?.releaseName,
               releaseNotes: update?.releaseNotes,
               releaseUrl: update?.releaseUrl,
               downloadable: update?.downloadable !== false,
             };
           case 'download-started':
           case 'download-progress': {
             const directProgress = clampProgress(payload.progress);
             const calculatedProgress =
               typeof payload.downloaded === 'number' &&
               typeof payload.total === 'number' &&
               Number.isFinite(payload.downloaded) &&
               Number.isFinite(payload.total) &&
               payload.total > 0 &&
               payload.downloaded >= 0
                 ? clampProgress(payload.downloaded / payload.total)
                 : undefined;
             const progress = directProgress ?? calculatedProgress;
             return {
               phase: 'downloading',
               releaseName: update?.releaseName,
               releaseNotes: update?.releaseNotes,
               releaseUrl: update?.releaseUrl,
               progress,
             };
           }
           case 'download-ready':
           case 'install-deferred':
           case 'install-preparing':
           case 'install-handoff':
             return {
               phase: 'ready',
               releaseName: update?.releaseName,
               releaseNotes: update?.releaseNotes,
               releaseUrl: update?.releaseUrl,
             };
           case 'download-failed':
             return {
               phase: 'error',
               releaseName: update?.releaseName,
               releaseNotes: update?.releaseNotes,
               releaseUrl: update?.releaseUrl,
               error: payload.error || payload.reason ? sanitizeUserPaths(String(payload.error || payload.reason)) : 'Update download failed',
             };
           default:
             return null;
         }
       }

       export type TranslationSyncStatus =
         | 'preparing'
         | 'downloading'
         | 'extracting-binaries'
         | 'ready'
         | 'error'
         | 'sync-extracting'
         | 'syncing'
         | 'synced'
         | 'sync-error'
         | 'sync-cancelled'
         | 'sync-queued';

       export interface TranslationQueueItem {
         gameKey: string;
         title?: string;
       }

       export interface TranslationSyncNotification {
         status: TranslationSyncStatus;
         gameKey?: string;
         progress?: number;
         queue?: TranslationQueueItem[];
         error?: string;
       }

       const VALID_TRANSLATION_STATUSES = new Set<TranslationSyncStatus>([
         'preparing', 'downloading', 'extracting-binaries', 'ready', 'error',
         'sync-extracting', 'syncing', 'synced', 'sync-error', 'sync-cancelled', 'sync-queued'
       ]);

       const DANGEROUS_KEYS = new Set(['__proto__', 'constructor', 'prototype']);

       export function parseTranslationStatusPayload(raw: unknown): TranslationSyncNotification | null {
         if (!raw || typeof raw !== 'object' || !('status' in raw)) return null;
         const payload = raw as Record<string, any>;
         if (!VALID_TRANSLATION_STATUSES.has(payload.status)) return null;

         const sanitizedQueue = Array.isArray(payload.queue)
           ? payload.queue
               .filter((item): item is Record<string, any> =>
                 Boolean(
                   item &&
                   typeof item === 'object' &&
                   typeof item.gameKey === 'string' &&
                   item.gameKey.trim().length > 0 &&
                   !DANGEROUS_KEYS.has(item.gameKey.trim())
                 )
               )
               .map((item) => ({
                 gameKey: String(item.gameKey).trim(),
                 title: typeof item.title === 'string' && item.title.trim().length > 0 ? item.title.trim() : undefined,
               }))
           : undefined;

         const parsedProgress = clampProgress(payload.progress);
         const progress =
           parsedProgress !== undefined
             ? parsedProgress
             : (payload.status === 'synced' || payload.status === 'ready' ? 1 : undefined);

         return {
           status: payload.status,
           gameKey: typeof payload.gameKey === 'string' ? payload.gameKey : undefined,
           progress,
           queue: sanitizedQueue,
           error: typeof payload.error === 'string' ? sanitizeUserPaths(payload.error) : undefined,
         };
       }

       export interface NotificationContextValue {
         showToast: (message: string, durationMs?: number) => void;
         activeUpdate: AppUpdateNotification | null;
         activeTranslation: TranslationSyncNotification | null;
         dismissUpdate: () => void;
         dismissTranslation: () => void;
         startUpdateDownload: () => Promise<void>;
         openUpdateDownloadPage: () => Promise<void>;
         restartAndInstallUpdate: () => Promise<void>;
         scheduleInstallOnNextLaunch: () => Promise<void>;
         cancelTranslationSync: (gameKey: string) => Promise<void>;
         moveTranslationQueue: (params: { gameKey: string; direction: 'up' | 'down' }) => Promise<void>;
         setFallbackFocusTarget: (target: HTMLElement | null) => void;
       }

       export function useNotification(): NotificationContextValue;
       ```

         ```tsx
         export function TranslationSyncBanner({
           translation,
           onDismiss,
           onCancelSync,
           fallbackFocusRef,
         }: TranslationSyncBannerProps) {
           const [isCancelling, setIsCancelling] = useState(false);
           const isError = translation.status === 'error' || translation.status === 'sync-error';
           const isProgress =
             typeof translation.progress === 'number' &&
             (translation.status === 'downloading' || translation.status === 'syncing');

           const focusFallbackTarget = () => {
             const target =
               fallbackFocusRef?.current ??
               document.querySelector<HTMLElement>('#react-root') ??
               document.querySelector<HTMLElement>('.library-grid');
             if (target) {
               if (!target.hasAttribute('tabindex')) {
                 target.setAttribute('tabindex', '-1');
               }
               target.focus();
             }
           };

           const handleDismiss = () => {
             if (isCancelling) return;
             focusFallbackTarget();
             onDismiss();
           };

           const handleCancel = async () => {
             if (isCancelling || !onCancelSync || !translation.gameKey) return;
             setIsCancelling(true);
             try {
               await onCancelSync(translation.gameKey);
               // Physical focus retention: shift focus to main application container ONLY upon successful cancellation resolution
               focusFallbackTarget();
             } catch (error) {
               const rawMsg = error instanceof Error ? error.message : String(error);
               const sanitizedMsg = sanitizeUserPaths(rawMsg);
               const sanitizedStack = error instanceof Error && error.stack ? sanitizeUserPaths(error.stack) : undefined;
               console.error('[TranslationSyncBanner] Failed to cancel translation sync:', {
                 gameKey: sanitizeUserPaths(translation.gameKey),
                 error: sanitizedMsg,
                 stack: sanitizedStack,
               });
               setIsCancelling(false);
               // Active focus remains safely on the Cancel button so user can retry or inspect error
             }
           };

           return (
             <aside
               className={`translation-sync-banner ${isError ? 'is-error' : ''}`}
               role="region"
               aria-label="Translation sync progress"
               aria-live="polite"
             >
               <div className="translation-sync-header">
                 <span className="translation-sync-title">
                   {isError ? 'Translation Error' : 'Translating Game Content'}
                 </span>
                 <button
                   type="button"
                   className="translation-sync-dismiss"
                   aria-label="Dismiss translation status"
                   title="Dismiss translation status"
                   aria-disabled={isCancelling}
                   style={isCancelling ? { pointerEvents: 'none' } : undefined}
                   onClick={handleDismiss}
                 >
                   <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" aria-hidden="true">
                     <path d="M18 6L6 18M6 6l12 12" />
                   </svg>
                 </button>
               </div>
               {isProgress && (
                 <div
                   className="translation-progress-track"
                   role="progressbar"
                   aria-valuenow={Math.round((translation.progress ?? 0) * 100)}
                   aria-valuemin={0}
                   aria-valuemax={100}
                 >
                   <div
                     className="translation-progress-bar"
                     style={{ width: `${Math.round((translation.progress ?? 0) * 100)}%` }}
                   />
                 </div>
               )}
               {translation.error && (
                 <p className="translation-sync-error">{translation.error}</p>
               )}
               {translation.gameKey && (translation.status === 'syncing' || translation.status === 'downloading') && onCancelSync && (
                 <div className="translation-sync-actions">
                   <button
                     type="button"
                     className="small-btn secondary-btn"
                     aria-disabled={isCancelling ? 'true' : undefined}
                     style={{
                       pointerEvents: isCancelling ? 'none' : undefined,
                       opacity: isCancelling ? 0.6 : 1,
                       cursor: isCancelling ? 'not-allowed' : 'pointer'
                     }}
                     onClick={handleCancel}
                   >
                     {isCancelling ? 'Cancelling...' : 'Cancel Sync'}
                   </button>
                 </div>
               )}
             </aside>
           );
         }
         ```
