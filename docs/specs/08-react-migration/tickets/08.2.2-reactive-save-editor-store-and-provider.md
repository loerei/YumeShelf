### [x] 2. **Actions, Session Isolation, Tab Indexing, Snapshot Immutability & $O(1)$ Dirty Tracking**:
   - Implement action methods on the store:
     - `setFileLoading(fileName: string)`:
       - Resets `this.modifiedFields = {}`, `this.dirtyKeys.clear()`, `this.searchQuery = ''`, sets `this.isTranslating = false`, re-instantiates fresh Map instances for `this.pristineFieldsById = new Map()` and `this.pristineFieldsByTab = new Map()` (never mutating in-place via `.clear()`), and sets `this.saveStatus = { state: 'idle' }`, isolating sessions across file switches and invalidating the snapshot.
       - Sets `this.fileStatus = { state: 'loading', fileName }`.
     - `setFileProgress(fileName: string, percent?: number)`:
       - Defensively clamps `percent` via `const clampedPercent = typeof percent === 'number' && Number.isFinite(percent) ? Math.max(0, Math.min(100, Math.round(percent))) : undefined;`.
       - Updates `this.fileStatus = { state: 'loading', fileName, percent: clampedPercent }` and invalidates snapshot.
     - `setFileLoaded(dto: NormalizedSaveDto)`:
       - Enforces Unicode NFC normalization on all field labels (`field.label`) at store ingress and symmetrically indexes both `pristineFieldsById` and `pristineFieldsByTab` upon initial load using canonical `field.tabId`:
         ```typescript
         const normalizedFields = (dto.fields || []).map((f) => ({
           ...f,
           label: (f.label || '').normalize('NFC'),
         }));
         this.pristineFieldsById = new Map(normalizedFields.map((f) => [f.id, f]));
         this.pristineFieldsByTab = new Map();
         for (const field of normalizedFields) {
           const tabId = field.tabId || 'general';
           const list = this.pristineFieldsByTab.get(tabId) || [];
           list.push(field);
           this.pristineFieldsByTab.set(tabId, list);
         }
         ```
       - Resets `this.modifiedFields = {}`, `this.dirtyKeys.clear()`, and `this.searchQuery = ''`, invalidating the snapshot.
       - Sets `this.activeTabId = dto.tabs[0]?.id || ''` if current `activeTabId` is invalid.
       - Sets `this.fileStatus = { state: 'loaded', dto: { ...dto, fields: normalizedFields } }` and `this.saveStatus = { state: 'idle' }`.
     - `setFileError(fileName: string, errorMessage: string)`:
       - Sets `this.fileStatus = { state: 'error', fileName, errorMessage }`.
     - `setFileIdle()`:
       - Sets `this.fileStatus = { state: 'idle' }`, `this.saveStatus = { state: 'idle' }`, `this.isTranslating = false`, `this.isMutatingFile = false`, `this.searchQuery = ''`, resets `this.modifiedFields = {}`, clears `this.dirtyKeys.clear()`, re-instantiates `this.pristineFieldsById = new Map()` and `this.pristineFieldsByTab = new Map()`, and invalidates snapshot via `this.invalidateSnapshot()`.
     - `renameActiveFile(newFileName: string, expectedCurrentName?: string)`:
       - When `this.fileStatus.state !== 'loaded'`, or when `!newFileName` or `typeof newFileName !== 'string'`, returns immediately.
       - When `expectedCurrentName` is provided, verifies `this.fileStatus.dto.fileName === expectedCurrentName.trim().normalize('NFC')`, returning immediately on mismatch.
       - Trims and normalizes `newFileName` via `const trimmed = newFileName.trim().normalize('NFC')`; if empty or identical to `this.fileStatus.dto.fileName`, returns immediately preserving referential snapshot stability.
       - Updates `this.fileStatus = { state: 'loaded', dto: { ...this.fileStatus.dto, fileName: trimmed } }` preserving `searchQuery`, `modifiedFields`, `dirtyKeys`, and pristine caches intact, and invalidates snapshot.
     - `setMutatingFile(isMutating: boolean)`:
       - Updates `this.isMutatingFile = Boolean(isMutating)` and invalidates snapshot if changed.
     - `setActiveTab(tabId: string)`:
       - Updates `this.activeTabId = tabId`.
     - `setSearchQuery(query: string)`:
       - Updates `this.searchQuery = query`.
     - `setTranslating(isTranslating: boolean)`:
       - Updates `this.isTranslating = isTranslating`.
     - `setTranslations(newTranslations: Record<string, string>)`:
       - Filters dangerous prototype pollution keys (`__proto__`, `constructor`, `prototype`), normalizes keys and string values to Unicode NFC ('NFC') at store ingress, and performs an additive dictionary merge rather than full replacement so translations persist across incremental batch chunks and tab switches:
         ```typescript
         const DANGEROUS_KEYS = new Set(['__proto__', 'constructor', 'prototype']);
         const clean: Record<string, string> = {};
         if (newTranslations && typeof newTranslations === 'object' && !Array.isArray(newTranslations)) {
           for (const [k, v] of Object.entries(newTranslations)) {
             if (!DANGEROUS_KEYS.has(k) && typeof v === 'string') {
               clean[k.normalize('NFC')] = v.normalize('NFC');
             }
           }
         }
         this.translations = { ...this.translations, ...clean };
         ```
     - `hydrateTranslations(translations: Record<string, string>)`:
       - Filters dangerous prototype pollution keys (`__proto__`, `constructor`, `prototype`), normalizes keys and string values to Unicode NFC ('NFC') at store ingress, and additively hydrates pre-loaded persisted translations into `this.translations`:
         ```typescript
         const DANGEROUS_KEYS = new Set(['__proto__', 'constructor', 'prototype']);
         const clean: Record<string, string> = {};
         if (translations && typeof translations === 'object' && !Array.isArray(translations)) {
           for (const [k, v] of Object.entries(translations)) {
             if (!DANGEROUS_KEYS.has(k) && typeof v === 'string') {
               clean[k.normalize('NFC')] = v.normalize('NFC');
             }
           }
         }
         this.translations = { ...this.translations, ...clean };
         ```
     - `updateField(fieldId: string, value: unknown)`:
       - Looks up `pristine = this.pristineFieldsById.get(fieldId)`. If not found, returns early.
       - Coerces scalar input type based on `pristine.type` to safeguard binary serialization integrity:
         - If `pristine.type === 'number'`: if `typeof value === 'string'`, when `value.trim() === ''`, retain empty string draft or return early (avoiding coercion to `0` or `NaN` during typing); otherwise coerce with `Number(value)` and reject/ignore if `!Number.isFinite(coerced)`. Normalize `-0` to `0` via `const normalizedVal = Object.is(coerced, -0) ? 0 : coerced;`.
         - If `pristine.type === 'boolean'`, coerce with `Boolean(value)` (or `value === 'true' || value === true`).
         - If `pristine.type === 'string'`, coerce with `String(value)`.
       - Compares normalized `value` against `pristine.value` in $O(1)$ time:
         - If identical to pristine (`Object.is(normalizedVal, pristine.value)`), deletes `fieldId` from `modifiedFields` and `dirtyKeys`.
         - If differing, sets `modifiedFields[fieldId] = normalizedVal` and `dirtyKeys.add(fieldId)`.
       - If `saveStatus.state === 'saved'`, transitions `saveStatus` back to `{ state: 'idle' }` to remove the stale indication.
     - `resetField(fieldId: string)`:
       - Deletes `fieldId` from `modifiedFields` and `dirtyKeys`.
     - `clearDirty()`:
       - Empties `modifiedFields` and clears `dirtyKeys`.
     - `setSaving()`:
       - Sets `saveStatus = { state: 'saving' }`.
     - `setSaved(committedUpdates: SaveUpdatesPayload, fileName: string)` (Race-Safe Baseline Commit):
       - Callers initiating a save must snapshot `const targetFileName = snapshot.fileStatus.dto.fileName` and `const inFlightUpdates = { ...snapshot.modifiedFields }` before invoking `electronAPI.writeSaveData({ gameKey, fileName: targetFileName, updates: inFlightUpdates })`, and pass `store.setSaved(inFlightUpdates, targetFileName)` on resolution.
       - Guards against non-loaded state or cross-file session switches during an in-flight save, reconciles `modifiedFields` against pre-commit effective UI values *before* overwriting `pristineFieldsById` (preserving both in-flight edits to new values and in-flight reversions to pre-save pristine values), updates only committed fields in `dto.fields`, and symmetrically re-indexes `pristineFieldsByTab` by `field.tabId`:
         ```typescript
         setSaved(committedUpdates: SaveUpdatesPayload, fileName: string): void {
           // 1. Guard against non-loaded state or cross-file session switch during in-flight save
           if (
             this.fileStatus.state !== 'loaded' ||
             this.fileStatus.dto.fileName !== fileName
           ) {
             return;
           }

           // 2. Reconcile modifiedFields against pre-commit effective UI values BEFORE overwriting pristineFieldsById.
           // Preserves both in-flight edits to new values AND in-flight reversions to the pre-save pristine value.
           for (const [key, committedVal] of Object.entries(committedUpdates)) {
             const prevPristine = this.pristineFieldsById.get(key);
             if (!prevPristine) continue;
             const currentEffectiveVal = this.dirtyKeys.has(key)
               ? this.modifiedFields[key]
               : prevPristine.value;

             if (Object.is(currentEffectiveVal, committedVal)) {
               delete this.modifiedFields[key];
               this.dirtyKeys.delete(key);
             } else {
               this.modifiedFields[key] = currentEffectiveVal;
               this.dirtyKeys.add(key);
             }
           }

           // 3. Only fields actually committed to disk are updated in the baseline DTO.
           const committedFields = this.fileStatus.dto.fields.map((field) =>
             Object.prototype.hasOwnProperty.call(committedUpdates, field.id)
               ? { ...field, value: committedUpdates[field.id] }
               : field
           );
           this.fileStatus = {
             state: 'loaded',
             dto: { ...this.fileStatus.dto, fields: committedFields }
           };
           this.pristineFieldsById = new Map(committedFields.map((f) => [f.id, f]));

           // 4. Symmetrically synchronize pristineFieldsByTab index using canonical field.tabId
           this.pristineFieldsByTab = new Map();
           for (const field of committedFields) {
             const tabId = field.tabId || 'general';
             const list = this.pristineFieldsByTab.get(tabId) || [];
             list.push(field);
             this.pristineFieldsByTab.set(tabId, list);
           }

           this.saveStatus = this.dirtyKeys.size === 0 ? { state: 'saved' } : { state: 'idle' };
           this.invalidateSnapshot();
         }
         ```
     - `setSaveError(errorMessage: string, fileName?: string)`:
       - If `fileName !== undefined && (this.fileStatus.state !== 'loaded' || this.fileStatus.dto.fileName !== fileName)`, returns immediately without mutating state, preventing cross-file error banner contamination during background writes.
       - Sets `saveStatus = { state: 'error', errorMessage }` and invalidates snapshot.

     ```tsx
     export interface SaveEditorStoreProviderProps {
       gameKey?: string;
       store?: ObservableSaveEditorStore;

       children: React.ReactNode;

     export function SaveEditorStoreProvider({
       gameKey,
       store,
       electronAPI,
       children,
     }: SaveEditorStoreProviderProps) {
       const resolvedStore = useMemo(() => store ?? createSaveEditorStore(gameKey ?? ''), [store, gameKey]);
       const dispatchers: SaveEditorActionDispatchers = useMemo(
         () => ({
           setFileLoading: (fileName) => resolvedStore.setFileLoading(fileName),
           setFileProgress: (fileName, percent) => resolvedStore.setFileProgress(fileName, percent),
           setFileLoaded: (dto) => resolvedStore.setFileLoaded(dto),
           setFileError: (fileName, errorMessage) => resolvedStore.setFileError(fileName, errorMessage),
           setFileIdle: () => resolvedStore.setFileIdle(),
           setActiveTab: (tabId) => resolvedStore.setActiveTab(tabId),
           setSearchQuery: (query) => resolvedStore.setSearchQuery(query),
           setTranslations: (translations) => resolvedStore.setTranslations(translations),
           hydrateTranslations: (translations) => resolvedStore.hydrateTranslations(translations),
           setTranslating: (isTranslating) => resolvedStore.setTranslating(isTranslating),
           setMutatingFile: (isMutating) => resolvedStore.setMutatingFile(isMutating),
           renameActiveFile: (newFileName, expectedCurrentName) => resolvedStore.renameActiveFile(newFileName, expectedCurrentName),
           updateField: (key, value) => resolvedStore.updateField(key, value),
           resetField: (key) => resolvedStore.resetField(key),
           clearDirty: () => resolvedStore.clearDirty(),
           setSaving: () => resolvedStore.setSaving(),
           setSaved: (committedUpdates, fileName) => resolvedStore.setSaved(committedUpdates, fileName),
           setSaveError: (errorMessage, fileName) => resolvedStore.setSaveError(errorMessage, fileName),
         }),
         [resolvedStore]
       );

       useEffect(() => {
         if (typeof window === 'undefined') return;
         const handleBeforeUnload = (e: BeforeUnloadEvent) => {
           if (resolvedStore.getSnapshot().dirtyKeys.size > 0) {
             e.preventDefault();
             e.returnValue = '';

         };
         window.addEventListener('beforeunload', handleBeforeUnload);
         return () => window.removeEventListener('beforeunload', handleBeforeUnload);
       }, [resolvedStore]);

         <SaveEditorElectronApiContext.Provider value={electronAPI}>
           <SaveEditorStoreContext.Provider value={resolvedStore}>
             <SaveEditorDispatchContext.Provider value={dispatchers}>
               {children}
             </SaveEditorDispatchContext.Provider>
           </SaveEditorStoreContext.Provider>
         </SaveEditorElectronApiContext.Provider>
       );
     }
     ```
