     - Number fields render `<NumberInput id={`field-${field.id}`} label={label} value={value} onChange={(val) => onChange(field.id, val)} />`, maintaining local string drafting state to allow intermediate negative numbers (`-`), intermediate decimal points (`.` or `12.`), and backspace clearing, committing validated finite numbers (including floating-point decimals) on blur or Enter without integer truncation, marking invalid/out-of-range values with `aria-invalid="true"` and linking inline validation messages via `aria-describedby`:
       ```tsx
       export interface NumberInputProps {
         id?: string;
         label?: string;
         value: number;
         onChange: (val: number) => void;
         disabled?: boolean;
       }

       export function NumberInput({
         id,
         label,
         value,
         onChange,
         disabled = false,
       }: NumberInputProps) {
         const [draft, setDraft] = useState(String(value));
         const [draftError, setDraftError] = useState<string | null>(null);
         const skipBlurCommitRef = useRef(false);

         useEffect(() => {
           skipBlurCommitRef.current = false;
           setDraft(String(value));
           setDraftError(null);
         }, [value]);

         const handleCommit = (fromEnter = false) => {
           const trimmed = draft.trim();
           if (trimmed === '' || trimmed === '-' || trimmed === '.' || trimmed === '-.') {
             skipBlurCommitRef.current = false;
             setDraft(String(value));
             setDraftError(null);
             return;
           }
           const parsed = Number(trimmed);
           if (Number.isFinite(parsed)) {
             skipBlurCommitRef.current = fromEnter;
             setDraft(String(parsed));
             setDraftError(null);
             if (parsed !== value) {
               onChange(parsed);
             }
           } else {
             skipBlurCommitRef.current = false;
             if (fromEnter) {
               setDraftError('Invalid number');
             } else {
               setDraft(String(value));
               setDraftError(null);
             }
           }
         };

         return (
           <div className="virtualized-table-number-input-wrapper">
             <input
               id={id}
               type="text"
               inputMode="decimal"
               className="virtualized-table-number-input"
               aria-label={label}
               aria-invalid={Boolean(draftError)}
               aria-describedby={draftError && id ? `${id}-error` : undefined}
               value={draft}
               disabled={disabled}
               onChange={(e) => {
                 skipBlurCommitRef.current = false;
                 setDraft(e.target.value);
                 setDraftError(null);
               }}
               onBlur={() => {
                 if (skipBlurCommitRef.current) {
                   skipBlurCommitRef.current = false;
                   return;
                 }
                 handleCommit(false);
               }}
               onKeyDown={(e) => {
                 if (e.key === 'Enter') {
                   handleCommit(true);
                 }
               }}
             />
             {draftError && id && (
               <span id={`${id}-error`} className="number-input-error" role="alert">
                 {draftError}
               </span>
             )}
           </div>
         );
       }
       ```
