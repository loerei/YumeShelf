When `isMenuOpen === true`, mount an encapsulated `useEffect` listener on `document` to dismiss the menu on outside `pointerdown` and `Escape` keydown, returning focus cleanly to `menuBtnRef.current?.focus()`:
           ```tsx
           const [isMenuOpen, setIsMenuOpen] = useState(false);
           const [isRenaming, setIsRenaming] = useState(false);
           const [draftName, setDraftName] = useState(game.name);

           const cardRef = useRef<HTMLDivElement | null>(null);
           const menuRef = useRef<HTMLDivElement | null>(null);
           const menuBtnRef = useRef<HTMLButtonElement | null>(null);
           const renameInputRef = useRef<HTMLInputElement | null>(null);
           const isCancellingRef = useRef(false);
           const isCommittingRef = useRef(false);

           // Merged ref handler for forwardRef composition
           const handleRootRef = (node: HTMLDivElement | null) => {
             cardRef.current = node;
             if (typeof forwardedRef === 'function') {
               forwardedRef(node);
             } else if (forwardedRef) {
               (forwardedRef as React.MutableRefObject<HTMLDivElement | null>).current = node;
             }
           };

           // Synchronize draftName when game.name updates externally
           useEffect(() => {
             setDraftName(game.name);
           }, [game.name]);

           useLayoutEffect(() => {
             if (isRenaming) {
               renameInputRef.current?.focus();
               renameInputRef.current?.select();
             }
           }, [isRenaming]);

           useEffect(() => {
             if (!isMenuOpen) return;

             const handleDocumentPointerDown = (e: PointerEvent) => {
               const target = e.target as Node | null;
               if (
                 menuRef.current &&
                 target &&
                 !menuRef.current.contains(target) &&
                 menuBtnRef.current &&
                 !menuBtnRef.current.contains(target)
               ) {
                 setIsMenuOpen(false);
               }
             };

             const handleDocumentKeyDown = (e: KeyboardEvent) => {
               if (e.key === 'Escape') {
                 e.preventDefault();
                 setIsMenuOpen(false);
                 menuBtnRef.current?.focus();
               }
             };

             document.addEventListener('pointerdown', handleDocumentPointerDown);
             document.addEventListener('keydown', handleDocumentKeyDown);
             return () => {
               document.removeEventListener('pointerdown', handleDocumentPointerDown);
               document.removeEventListener('keydown', handleDocumentKeyDown);
             };
           }, [isMenuOpen]);

           const handleStartRename = () => {
             setIsMenuOpen(false);
             isCancellingRef.current = false;
             isCommittingRef.current = false;
             setDraftName(game.name);
             setIsRenaming(true);
           };

           const handleRenameCommit = (options?: { restoreFocus?: boolean }) => {
             if (isCancellingRef.current || isCommittingRef.current) return;
             isCommittingRef.current = true;
             const trimmed = draftName.trim();
             if (trimmed && trimmed !== game.name) {
               onMenuAction?.('rename', game, { newName: trimmed });
             } else {
               setDraftName(game.name);
             }
             setIsRenaming(false);
             if (options?.restoreFocus ?? true) {
               cardRef.current?.focus();
             }
           };

           const handleRenameCancel = () => {
             if (isCancellingRef.current || isCommittingRef.current) return;
             isCancellingRef.current = true;
             setDraftName(game.name);
             setIsRenaming(false);
             menuBtnRef.current?.focus();
           };

           const handleMenuSelect = (action: GameCardMenuAction, payload?: GameCardMenuPayload) => {
             menuBtnRef.current?.focus();
             setIsMenuOpen(false);
             onMenuAction?.(action, game, payload);
           };
           ```
