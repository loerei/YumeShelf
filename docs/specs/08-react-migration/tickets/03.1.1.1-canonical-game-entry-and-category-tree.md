2. **Canonical Category Tree Domain Module (`src/renderer/domain/category-tree.ts`)**:
   - Consolidate the canonical recursive category hierarchy node (`CategoryNode`), flattened presentation item (`FlattenedCategoryItem`), and pure tree traversal algorithm (`flattenCategoryTree`) in `src/renderer/domain/category-tree.ts`:
     ```typescript
     export interface CategoryNode {
       id: string;
       name: string;
       parentId?: string | null;
       children?: CategoryNode[];
     }

     export interface FlattenedCategoryItem {
       id: string;
       name: string;
       label?: string;
       depth: number;
       trailLabel: string;
     }

     export function flattenCategoryTree(
       tree: readonly CategoryNode[],
       depth = 0,
       parentNames: string[] = [],
       visited: Set<string> = new Set<string>()
     ): FlattenedCategoryItem[] {
       const result: FlattenedCategoryItem[] = [];
       if (depth > 32) return result;
       for (const node of Array.isArray(tree) ? tree : []) {
         if (!node || !node.id || visited.has(node.id)) continue;
         const nodeName = typeof node?.name === 'string' ? node.name.trim() : '';
         if (!nodeName) continue;
         visited.add(node.id);
         const trail = [...parentNames, nodeName];
         result.push({
           id: node.id,
           name: nodeName,
           label: nodeName,
           depth,
           trailLabel: trail.join(' / ')
         });
         if (Array.isArray(node.children) && node.children.length > 0) {
           result.push(...flattenCategoryTree(node.children, depth + 1, trail, visited));
         }
       }
       return result;
     }
     ```

     ```typescript
     export type LocalizationDictionary = Record<string, string | undefined>;
     export type LocalizationProvider = (() => LocalizationDictionary) | LocalizationDictionary;

     export function timeSince(
       date: Date | string | number | null | undefined,
       getStrings: LocalizationProvider,
       now: Date | number = new Date()
     ): string {
       const d = typeof getStrings === 'function' ? getStrings() : (getStrings || {});
       if (!date || date === 0) return d.status_never || '';

       const past = new Date(date);
       if (isNaN(past.getTime())) return '';

       const current = typeof now === 'number' ? new Date(now) : (now instanceof Date ? now : new Date(now));
       if (isNaN(current.getTime())) return '';

       const diffMs = current.getTime() - past.getTime();
       if (diffMs < 0) return d.status_recent || ''; // Clock skew / future timestamp guard

       const seconds = Math.floor(diffMs / 1000);
       if (seconds < 60) return d.status_recent || '';

       const minutes = Math.floor(seconds / 60);
       if (minutes < 60) return `${minutes}${d.status_mins || ''}`;

       const hours = Math.floor(seconds / 3600);
       if (hours < 24) return `${hours}${d.status_hours || ''}`;

       const days = Math.floor(seconds / 86400);
       if (days < 30) return `${days}${d.status_days || ''}`;

       const months = Math.floor(seconds / (86400 * 30));
       if (months < 12) return `${months}${d.status_months || ''}`;

       const years = Math.floor(seconds / (86400 * 365));
       return `${years}${d.status_years || ''}`;
     }
     ```
