guaranteeing crash-free execution in headless Vitest environments. Export pure function seam `createGameLibraryPutValidator(context)` (and expose as `groupPutValidator` on the controller instance) implementing the 5-scenario drop permission matrix: (1) Same-container reordering allowed (`to.el === from.el`), (2) Flat sort modes (`date`, `played`, `az`, `custom`) allow cross-zone drags between `favGrid` and `unfavGrid`, (3) Folder sort into favorites allowed only if card is not already present in favorites, (4) Folder sort out of favorites allowed only into owning root folder container (matching `targetRoot` with platform normalization or `isSubsumedBy`, returning `false` immediately if `targetRoot` from `to.el.dataset.rootFolderPath` is undefined or non-string to prevent `TypeError` in `normalizePathForPlatform`), (5) Cross-folder drops between disparate folder sections strictly rejected.

In `handleCardDragEnd(evt)`, resolve adjacent sibling references (`evt.item.nextElementSibling` or `evt.item.previousElementSibling`) and calculate relative insertion into `customOrder` *before* DOM mutation. In `folder` sort mode, calculate insertion into global `customOrder` relative to adjacent sibling games within destination folder container rather than local container index. Reconcile favorite state across zones (`applyFavoriteToLogicalGame` and `electronAPI.toggleFavorite`), auto-switch to `custom` sort only when not in `folder` mode, update `customOrder` including grouped keys via `getGroupedKeysForGame`, persist via `writeCustomOrder`, and defer reactive view refresh via `scheduleRefresh(() => options.sortGames?.(getCurrentSort()))`. Implement public lifecycle methods: `bindGrids(containers: HTMLElement[])` (safely destroying pre-existing Sortable instances on stale containers), `destroy()` (tearing down all active Sortable instances), and `updateDisabled(disabled: boolean)`. To maintain runtime backward compatibility with `app-composition.ts` prior to cutover in Ticket `01.3.2`, `createDragDropGridController` returns transitional no-op stubs (`attachZoneHandlers: () => void`, `startDrag: (gameKey: string) => boolean`, `resetDragState: () => void`) alongside `bindGrids`, `destroy`, and `updateDisabled`.

Delete obsolete legacy math files `src/renderer/utils/drag-math.ts` and `src/renderer/utils/flip-animation.ts`. Provide comprehensive Vitest unit tests in `src/renderer/drag-drop-grid.test.ts`.

    ```typescript
    export function createGameLibraryPutValidator(context: {
      getCurrentSort: () => string;
      getFavGrid: () => HTMLElement | null;
      targetPlatform?: PlatformInput;
      isReordering?: () => boolean;
    }) {
      return (to: { el: HTMLElement }, from: { el: HTMLElement }, dragEl: HTMLElement): boolean => {
        // Block all cross-container drops while folder section reordering is in flight
        if (context.isReordering?.()) return false;
        // Early rejection guard: reject drops into collapsed or inert destination containers
        if (to.el.closest('.library-section')?.classList.contains('is-collapsed') || to.el.hasAttribute('inert')) return false;
        // Strict boundary check: elements lacking a valid, non-empty dataset.gameKey are strictly rejected
        if (!dragEl.dataset?.gameKey || typeof dragEl.dataset.gameKey !== 'string' || !dragEl.dataset.gameKey.trim()) return false;
        // Same container reordering is always permitted (including internal reordering within favGrid)
        if (to.el === from.el) return true;
        // Non-game elements (e.g. mascot game cards) are strictly excluded from cross-container drops
        if (dragEl.classList.contains('mascot-game-card') || dragEl.dataset.gameKey === '__mascot_card__') return false;
        // In flat sort modes (date, played, az, custom), cross-zone drag between favGrid and unfavGrid is unconditionally allowed
        if (context.getCurrentSort() !== 'folder') return true;
        // In folder sort mode:
        // Dragging into favorites grid: evaluate favorites grid identity via element ID or reference; allow only if card is not already in favorites
        const favGrid = context.getFavGrid();
        if (to.el.id === 'fav-grid' || to.el === favGrid) {
          const gameKey = dragEl.dataset.gameKey;
          return !to.el.querySelector('.game-card[data-game-key="' + safeEscapePath(gameKey) + '"]:not(.mascot-game-card)');
        }
        // Dragging from favorites grid back to folder sections: allow ONLY into owning root folder grid (checking targetRoot from to.el.dataset.rootFolderPath with platform normalization)
        if (from.el.id === 'fav-grid' || from.el === favGrid) {
          const cardRoot = dragEl.dataset.rootFolderPath;
          const targetRoot = to.el.dataset.rootFolderPath;
          if (!targetRoot || typeof targetRoot !== 'string') return false;
          return cardRoot
            ? normalizePathForPlatform(cardRoot, context.targetPlatform) === normalizePathForPlatform(targetRoot, context.targetPlatform)
            : isSubsumedBy(dragEl.dataset.folderPath || '', targetRoot, context.targetPlatform);
        }
        // Between standard folder grids: disallow moves across different folder grids
        return false;
      };
    }

    export function safeEscapePath(val: string): string {
      return typeof CSS !== 'undefined' && typeof CSS.escape === 'function'
        ? CSS.escape(val)
        : val.replace(/(["\\])/g, '\\$1');
    }
    ```
