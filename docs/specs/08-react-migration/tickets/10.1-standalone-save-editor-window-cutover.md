        export function SaveEditorApp({ gameKey, initialFile, store, electronAPI, onClose }: SaveEditorAppProps) {
         let queryGameKey = '';
         if (typeof window !== 'undefined') {
           try {
             queryGameKey = new URLSearchParams(window.location.search).get('gameKey')?.trim() ?? '';
           } catch {
             queryGameKey = '';
           }
         }
         const resolvedGameKey = gameKey?.trim() || queryGameKey;

         if (!resolvedGameKey && !store) {
           return (
             <div className="save-editor-error-banner" role="alert">
               Missing required gameKey parameter
             </div>
           );
         }

         const handleClose = onClose ?? (() => {
           if (typeof window !== 'undefined') {
             window.close();
           }
         });

         return (
            <AppErrorBoundary electronAPI={electronAPI}>
              <SaveEditorStoreProvider gameKey={resolvedGameKey} store={store} electronAPI={electronAPI}>
                <SaveEditorShell onClose={handleClose} initialFile={initialFile} />
              </SaveEditorStoreProvider>
            </AppErrorBoundary>
          );
       }

         onClose: () => void;
         initialFile?: string;
       }

       export function SaveEditorShell({ onClose, initialFile }: SaveEditorShellProps) {
         const retryCounterRef = useRef(0);
         const sidebarListFocusRef = useRef<HTMLDivElement | null>(null);
         const [retryRequest, setRetryRequest] = useState<{ fileName: string; requestId: number } | null>(null);

         const handleRetryLoad = (fileName: string) => {
           setRetryRequest({ fileName, requestId: ++retryCounterRef.current });
         };

         const handleChangeFile = () => {
           sidebarListFocusRef.current?.focus();
         };

         return (
           <div className="save-editor-overlay">
             <div className="save-editor-panel">
               <div className="save-editor-body">
                 <SaveEditorSidebar
                   initialFile={initialFile}
                   retryRequest={retryRequest}
                   listFocusRef={sidebarListFocusRef}
                 />

                   <div className="save-editor-top-bar">
                     <SaveEditorTabs />
                     <SaveEditorSearchBar />
                   </div>

                     onRetryLoad={handleRetryLoad}
                     onChangeFile={handleChangeFile}
                   />
                 </main>
               </div>
               <SaveEditorFooter />
             </div>
           </div>
         );
       }

       ```typescript
       import React from 'react';
       import { createRoot, type Root } from 'react-dom/client';
       import { SaveEditorApp } from './renderer/save-editor/SaveEditorApp';
       import type { ObservableSaveEditorStore } from './renderer/save-editor/store/save-editor-store';
       import type { ElectronAPI } from './shared/types/ipc';
       import { registerGlobalErrorSanitizer, sanitizeUserPaths } from './shared/path-sanitizer';

       declare global {
         interface Window {
           __IS_STANDALONE_SAVE_EDITOR__?: boolean;
           __VITEST__?: boolean;
         }
       }

       export interface StandaloneMountOptions {
         electronAPI?: ElectronAPI;
         store?: ObservableSaveEditorStore;
         controllerFactory?: () => unknown;
       }

       export async function mountStandaloneSaveEditor(
         root: HTMLElement | null = document.getElementById('save-editor-root'),
         search: string = window.location.search,
         options: StandaloneMountOptions = {}
       ): Promise<() => void> {
         if (!root) throw new Error('Failed to find #save-editor-root container');
         window.__IS_STANDALONE_SAVE_EDITOR__ = true;
         const bodyHadClass = document.body.classList.contains('standalone-save-editor');
         document.body.classList.add('standalone-save-editor');
         const unbindSanitizer = registerGlobalErrorSanitizer(window, '[SaveEditor]');

         const urlParams = new URLSearchParams(search);
         const gameKey = urlParams.get('gameKey')?.trim() || undefined;
         const initialFile = urlParams.get('file')?.trim() || undefined;

         let reactRoot: Root | null = createRoot(root);
         reactRoot.render(
           <SaveEditorApp
             gameKey={gameKey}
             initialFile={initialFile}
             store={options.store}
             electronAPI={options.electronAPI}
           />
         );

         let disposed = false;
         const cleanup = () => {
           if (disposed) return;
           disposed = true;
           window.removeEventListener('beforeunload', cleanup);
           unbindSanitizer();
           if (!bodyHadClass) {
             document.body.classList.remove('standalone-save-editor');
           }
           if (reactRoot) {
             reactRoot.unmount();
             reactRoot = null;
           }
         };

         window.addEventListener('beforeunload', cleanup);
         return cleanup;
       }

       if (typeof window !== 'undefined' && !window.__VITEST__) {
         void mountStandaloneSaveEditor().catch((err: unknown) => {
           const rawMsg = err instanceof Error ? (err.stack || err.message) : String(err);
           console.error('[SaveEditor] Failed to mount standalone save editor:', {
             error: sanitizeUserPaths(rawMsg),
           });
         });
       }
       ```
