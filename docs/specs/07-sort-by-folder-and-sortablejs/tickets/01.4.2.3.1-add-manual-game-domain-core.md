4. macOS bundle resolution:

Non-bundle executables (Mach-O binaries, ELF binaries, `.sh` scripts) fall through to standard containing directory resolution and executable inspection.
5. Binary inspection & owning root selection:
   - Resolves bundle root, directory path, and executable binary deterministically with pre-normalization and bounded wrapper unwrapping:
     ```typescript
     const normalizedTarget = normalizePathForPlatform(targetPath, platform);
     const isMacBundle = targetPath.toLowerCase().endsWith('.app') || Boolean(resolveBundleRoot(targetPath));
     let folderPath: string;
     if (isMacBundle) {
       folderPath = normalizePathForPlatform(resolveBundleRoot(targetPath) || targetPath, platform);
     } else {
       let currentDir = normalizedTarget.includes('/') ? normalizedTarget.slice(0, normalizedTarget.lastIndexOf('/')) || '/' : normalizedTarget;
       while (currentDir && currentDir !== '/') {
         const leafDir = currentDir.includes('/') ? currentDir.slice(currentDir.lastIndexOf('/') + 1) : currentDir;
         if (WRAPPER_DIRECTORY_NAMES.has(leafDir.toLowerCase())) {
           const parentDir = currentDir.includes('/') ? currentDir.slice(0, currentDir.lastIndexOf('/')) || '/' : '/';
           if (candidateRoots.some((root: string) => normalizePathForPlatform(parentDir, platform) === normalizePathForPlatform(root, platform)) ||
               !candidateRoots.some((root: string) => isSubsumedBy(parentDir, root, platform))) {
             break;
           }
           currentDir = parentDir;
         } else {
           break;
         }
       }
       folderPath = currentDir;
     }
     // If enclosingFolderPath is provided, validate containment strictly without overriding owningLibPath
     if (opts?.enclosingFolderPath) {
       const normalizedEnclosing = normalizePathForPlatform(opts.enclosingFolderPath, platform);
       if (!isSubsumedBy(normalizedTarget, normalizedEnclosing, platform)) {
         console.warn('[SECURITY][MANUAL_ADD] Blocked executable outside enclosing folder:', { targetPath, enclosingFolderPath: opts.enclosingFolderPath });
         return { ok: false, error: 'outside-enclosing-folder' };
       }
     }
      let exePath: string;
      if (isMacBundle) {
        const bundleExecutable = (await AppBundleInspector.fromPath(folderPath, adaptedFs))?.executablePath;
        if (bundleExecutable) {
          exePath = normalizePathForPlatform(bundleExecutable, platform);
        } else {
          const bundleName = (folderPath.replace(/\/+$/, '').split('/').pop()?.replace(/\.app$/i, '') || '').trim();
          if (!bundleName) {
            console.warn('[SECURITY][MANUAL_ADD] Empty bundle name in macOS app bundle fallback executable resolution:', { folderPath });
            return { ok: false, error: 'unresolvable-executable' };
          }
          const fallbackExe = normalizePathForPlatform(`${folderPath}/Contents/MacOS/${bundleName}`, platform);
          if (!isSubsumedBy(fallbackExe, normalizePathForPlatform(`${folderPath}/Contents/MacOS`, platform), platform)) {
            console.warn('[SECURITY][MANUAL_ADD] Blocked traversal in macOS app bundle fallback executable:', { folderPath, fallbackExe });
            return { ok: false, error: 'unresolvable-executable' };
          }
          const isRegularFile = await (context.fs?.stat
            ? context.fs.stat(fallbackExe).then(stat => Boolean(stat?.isFile?.()), () => false)
            : Promise.resolve(Boolean(context.fsSync?.statSync?.(fallbackExe)?.isFile?.())));
          if (!isRegularFile) {
            console.warn('[SECURITY][MANUAL_ADD] Failed to resolve regular executable binary for macOS app bundle on disk:', { folderPath, fallbackExe });
            return { ok: false, error: 'unresolvable-executable' };
          }
          exePath = fallbackExe;
        }
      } else {
        exePath = normalizedTarget;
      }
      ```
    - Matches owning library path strictly against `candidateRoots = subsumeLibraryPaths(config.libraryPaths, platform)`:
      ```typescript
      const owningLibPath = candidateRoots.find((root: string) => isSubsumedBy(folderPath, root, platform));
      if (!owningLibPath) {
        console.warn('[SECURITY][MANUAL_ADD] Folder path falls outside candidate library roots:', { folderPath, candidateRoots });
        return { ok: false, error: 'outside-library' };
      }
      if (normalizePathForPlatform(folderPath, platform) === normalizePathForPlatform(owningLibPath, platform)) {
        console.warn('[SECURITY][MANUAL_ADD] Blocked root-level executable without dedicated game folder:', { targetPath, folderPath, owningLibPath });
        return { ok: false, error: 'root-level-executable' };
      }
      ```
      ensuring `opts.enclosingFolderPath` is strictly a containment validator, root-level executables without a dedicated subdirectory are rejected to prevent key collision and deduplication purge, and never causes key divergence against `loader.ts`.
   - Derives `folderName = getFolderBaseName(folderPath)`, `gameKey = buildGameKey(owningLibPath, folderPath)`, and `relativePath = gameKey`.
   - Prototype security validation: guards `gameKey` against prototype pollution before database lookups or key assignments:

     ```typescript
     const isUnsafeGameKey = typeof gameKey !== 'string' ||
     ```

6. Database persistence phase dispatched through `context.queue(persistTask)`:
   - Degraded database guard: if `context.isDegraded?.() === true`, safely aborts and returns `{ ok: false, error: 'degraded-database' }`.
   - Authoritative snapshot re-query to eliminate lost update race hazards: inside `persistTask`, immediately re-query the latest database state under the queue lock:
     ```typescript
     const latestDb = await context.loadDB();
     const latestRoots = subsumeLibraryPaths(latestDb.config?.libraryPaths || [], platform);
     const latestOwningLibPath = latestRoots.find((root: string) => isSubsumedBy(folderPath, root, platform));
     if (!latestOwningLibPath) {
       console.warn('[SECURITY][MANUAL_ADD] Owning library path removed during binary inspection, aborting persistence:', { folderPath, latestRoots });
       return { ok: false, error: 'outside-library' };
     }
     if (normalizePathForPlatform(folderPath, platform) === normalizePathForPlatform(latestOwningLibPath, platform)) {
       console.warn('[SECURITY][MANUAL_ADD] Blocked root-level executable without dedicated game folder:', { targetPath, folderPath, latestOwningLibPath });
       return { ok: false, error: 'root-level-executable' };
     }
     const canonicalGameKey = buildGameKey(latestOwningLibPath, folderPath);
     const isUnsafeGameKey = typeof canonicalGameKey !== 'string' ||
       canonicalGameKey === '__proto__' ||
       canonicalGameKey === 'constructor' ||
       canonicalGameKey === 'prototype';
     if (isUnsafeGameKey) {
       console.warn('[SECURITY][MANUAL_ADD] Blocked unsafe game key:', { canonicalGameKey, folderPath });
       return { ok: false, error: 'invalid-game-key' };

     const games = readStoredGames(latestDb);
     ```
