## What to build
Mount the React DOM root container and wire the reactive bridge connecting the observable state store to React 19's `useSyncExternalStore` through a pure headless React Context:

1. **HTML Mounting Root Container (`src/index.html`)**:
   - Update [`src/index.html`](file:///d:/Projects/YumeShelf/src/index.html) to provide `<div id="react-root" tabindex="-1"></div>` and `<div id="portal-root"></div>` mounting containers alongside existing window layout containers, equipping `#react-root` with `tabindex="-1"` to support programmatic focus restoration without focus eviction to `document.body`.

2. **React Context & State Bridge Provider (`src/renderer/context/LibraryStateContext.tsx`)**:
   - Implement `LibraryStoreContext`, `LibraryDispatchContext`, and `<LibraryStateProvider store={store}>`:
     - Expose an injectable `store?: ObservableUiRuntimeStore` parameter seam. If no store is passed as a prop, fall back to default `createUiRuntimeState()` memoized via `useMemo(() => store ?? createUiRuntimeState(), [store])` (or `useRef`), guaranteeing referential stability across re-renders.
     - Segregate store reference and action dispatchers into two separate contexts:
       - `LibraryStoreContext`: Defined as `createContext<ObservableUiRuntimeStore | null>(null)`. Exposes the store reference directly (`value={resolvedStore}`). Referentially invariant across the entire application lifecycle, causing 0 context consumer re-renders.
       - `LibraryDispatchContext`: Defined as `createContext<LibraryActionDispatchers | null>(null)`. Exposes memoized action dispatchers (`bindDispatchers(resolvedStore)`).
       - Provider nesting:
         ```tsx
         <LibraryStoreContext.Provider value={resolvedStore}>
           <LibraryDispatchContext.Provider value={dispatchers}>
             {children}
           </LibraryDispatchContext.Provider>
         </LibraryStoreContext.Provider>
         ```
     - Action dispatchers on `LibraryActionDispatchers`: `setAllGames`, `updateGame`, `setGameRunning`, `updateGamePlaytime`, `setCategoryTree`, `setActiveCategoryId`, `setCurrentSort`, `setCurrentLibraryConfig`.
     - **Boundary Invariant**: Ticket 03.2 remains a pure headless state context bridge with zero knowledge of Electron IPC or startup cold-boot lifecycle (cold-boot orchestration belongs in Ticket 05).

3. **Consumer Hooks & Fine-Grained Selectors (`src/renderer/hooks/useLibraryState.ts`)**:
   - Implement fine-grained selector hooks powered by `useSyncExternalStoreWithSelector` to eliminate re-render cascades during high-frequency playtime updates (`onGamePlaytimeUpdated`):
     ```typescript
     import { useContext } from 'react';
     import { useSyncExternalStoreWithSelector } from 'use-sync-external-store/with-selector';
     import type { ObservableUiRuntimeStore, UiRuntimeSnapshot, SearchableGameItem } from '../state/ui-runtime-state';
     import type { CategoryNode } from '../domain/category-tree';
     import type { GameEntry, LibraryConfig } from '../types/game';
     import { LibraryStoreContext, LibraryDispatchContext } from '../context/LibraryStateContext';

     export function useLibraryStore(): ObservableUiRuntimeStore {
       const store = useContext(LibraryStoreContext);
       if (!store) throw new Error('useLibraryStore must be used within LibraryStateProvider');
       return store;
     }

     export function useLibrarySelector<T>(
       selector: (snapshot: UiRuntimeSnapshot) => T,
       isEqual: (a: T, b: T) => boolean = Object.is
     ): T {
       const store = useLibraryStore();
       return useSyncExternalStoreWithSelector(
         store.subscribe,
         store.getSnapshot,
         store.getSnapshot,
         selector,
         isEqual
       );
     }

     export function useActiveCategoryId(): string | null {
       return useLibrarySelector((s) => s.activeCategoryId);
     }

     export function useCurrentSort(): string {
       return useLibrarySelector((s) => s.currentSort);
     }

     export function useCategoryTree(): readonly CategoryNode[] {
       return useLibrarySelector((s) => s.categoryTree);
     }

     export function useAllGames(): readonly GameEntry[] {
       return useLibrarySelector((s) => s.allGames);
     }

     import { selectSearchableGames } from '../state/ui-runtime-state';
     export { selectSearchableGames };

     export function areSearchableGamesEqual(
       prev: readonly SearchableGameItem[],
       next: readonly SearchableGameItem[]
     ): boolean {
       if (Object.is(prev, next)) return true;
       if (!prev || !next || prev.length !== next.length) return false;
       for (let i = 0; i < prev.length; i++) {
         const p = prev[i];
         const n = next[i];
         if (p.gameKey !== n.gameKey || p.name !== n.name || p.iconUrl !== n.iconUrl) {
           return false;
         }
       }
       return true;
     }

     export function useSearchableGames(): readonly SearchableGameItem[] {
       return useLibrarySelector(selectSearchableGames, areSearchableGamesEqual);
     }

     export function useLocationDisplayMode(): 'parent' | 'full' {
       return useLibrarySelector((s) => s.locationDisplayMode);
     }

     export function useLibraryDispatch(): LibraryActionDispatchers {
       const dispatch = useContext(LibraryDispatchContext);
       if (!dispatch) throw new Error('useLibraryDispatch must be used within LibraryStateProvider');
       return dispatch;
     }

     export interface LibraryActionDispatchers {
       setAllGames(games: GameEntry[]): void;
       updateGame(gameKey: string, patch: Partial<GameEntry>): void;
       setGameRunning(gameKey: string, isRunning: boolean): void;
       updateGamePlaytime(gameKey: string, accruedMs: number): void;
       setCategoryTree(tree: CategoryNode[]): void;
       setActiveCategoryId(id: string | null): void;
       setCurrentSort(sort: string): void;
       setCurrentLibraryConfig(config: LibraryConfig | null): void;
       setLocationDisplayMode(mode: 'parent' | 'full'): void;
     }
     ```
   - Provide transitional `useLibraryState()` that aggregates the snapshot and dispatchers for Phase 1 components that have not yet migrated to granular selectors.
