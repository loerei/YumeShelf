    export interface UseLibraryActionsOptions {
      readonly store: ObservableUiRuntimeStore;
      readonly electronAPI?: ElectronAPI;
      readonly launcher?: ISaveEditorLauncher;

  - Extract `handleMenuAction`, `handleToggleFavorite`, `handleFavoriteError`, `handleLaunch`, `handleOpenSaveEditor`, `handleConfirmDelete`, `handleCloseDeleteModal`, and `isDeletingGame` out of `AppShell` (`App.tsx`) into `useLibraryActions.ts`, plus export a pure helper `recalculateAfterGameDeletion(...)` for direct isolated unit testing of stack reduction, dissolution, category filtering, and neighbor focus selection.

  - Keep `App.tsx` focused on top-level layout composition and provider wiring.
